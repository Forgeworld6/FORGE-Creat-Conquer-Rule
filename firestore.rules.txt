rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Each signed-in user can read and update only their own profile.
    match /users/{userId} {
      allow create: if request.auth != null
                    && request.auth.uid == userId
                    && request.resource.data.keys().hasOnly([
                      'name', 'bio', 'skills', 'phoneNumber',
                      'createdAt', 'updatedAt'
                    ]);
      allow read: if request.auth != null && request.auth.uid == userId;
      allow update: if request.auth != null
                    && request.auth.uid == userId
                    && request.resource.data.diff(resource.data).affectedKeys()
                       .hasOnly(['name', 'bio', 'skills', 'phoneNumber', 'updatedAt']);
      allow delete: if false;
    }

    // Everything else remains closed until its feature and rules are implemented.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
